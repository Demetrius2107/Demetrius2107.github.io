{{- /* 计算给定 time.Time 在当年的第几天（1-366），用于首页每日一句轮换 */ -}} {{- $t := . -}} {{- $year := $t.Year -}} {{- $day := $t.Day -}} {{- $months := dict "January" 1 "February" 2 "March" 3 "April" 4 "May" 5 "June" 6 "July" 7 "August" 8 "September" 9 "October" 10 "November" 11 "December" 12 -}} {{- $m := index $months (string $t.Month) -}} {{- /* 各月天数，下标 0=1月 … 11=12月；闰年二月单独处理 */ -}} {{- $daysInMonth := slice 31 28 31 30 31 30 31 31 30 31 30 31 -}} {{- $isLeap := and (eq (mod $year 4) 0) (or (ne (mod $year 100) 0) (eq (mod $year 400) 0)) -}} {{- $doy := 0 -}} {{- range $i := seq 1 (sub $m 1) -}} {{- $dm := index $daysInMonth (sub $i 1) -}} {{- if and (eq $i 2) $isLeap -}}{{- $dm = 29 -}}{{- end -}} {{- $doy = add $doy $dm -}} {{- end -}} {{- $doy = add $doy $day -}} {{- return $doy -}}
